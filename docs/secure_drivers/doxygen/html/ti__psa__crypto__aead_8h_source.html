48  /* Buffer for data that has not been processed yet. Word-aligned for max
49  * performance and in case any drivers require aligned input buffer.
50  * Double-buffer is used when PSA crypto is built into the TFM to prevent
51  * corruption before the data can be consumed since the underlying crypto
52  * drivers are used in callback mode. */
53 #ifdef TFM_BUILD
54  uint8_t unprocessedData[PSA_BLOCK_CIPHER_BLOCK_MAX_SIZE * 2] __attribute__((aligned(4)));
55 #else
56  uint8_t unprocessedData[PSA_BLOCK_CIPHER_BLOCK_MAX_SIZE] __attribute__((aligned(4)));
57 #endif /* TFM_BUILD */
58 
59  /* Pointer to the current unprocessed data */
60  uint8_t *curr_unprocessed_data;
61 
62  size_t unprocessed_len;
63 
64  CryptoKey cryptoKey;
65  bool done_updating_ad;
66  bool length_set;
67 };
